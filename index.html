Compare the same MRV backtracking idea in JavaScript, C++, C, and Java.

function solveMRV(board) {
  if (board.isComplete()) return true;
  const cell = board.emptyCellWithFewestCandidates();
  for (const digit of cell.candidates) {
    board.place(cell, digit);
    if (solveMRV(board)) return true;
    board.remove(cell); // backtrack
  }
  return false;
}
#include <bit>
#include <cstdint>

bool solve_mrv(Board& board) {
    if (board.complete()) return true;
    Cell cell = board.fewest_candidates();
    uint16_t mask = board.candidates(cell);
    while (mask) {
        int digit = std::countr_zero(mask) + 1;
        board.place(cell, digit);
        if (solve_mrv(board)) return true;
        board.remove(cell); // backtrack
        mask &= mask - 1;
    }
    return false;
}
#include <stdbool.h>
#include <stdint.h>

static int first_digit(uint16_t mask) {
    /* The lowest set bit represents the next digit to try. */
    int digit = 1;
    while ((mask & 1u) == 0u) {
        mask >>= 1;
        digit++;
    }
    return digit;
}

bool solve_mrv(Board *board) {
    if (board_complete(board)) return true;
    Cell cell = board_fewest_candidates(board);
    uint16_t mask = board_candidates(board, cell);

    while (mask != 0u) {
        int digit = first_digit(mask);
        board_place(board, cell, digit);
        if (solve_mrv(board)) return true;
        board_remove(board, cell); /* backtrack */
        /* Clear the candidate just tried, preserving the others. */
        mask &= (uint16_t)(mask - 1u);
    }
    return false;
}
boolean solveMrv(Board board) {
    if (board.isComplete()) return true;
    Cell cell = board.emptyCellWithFewestCandidates();
    int mask = board.candidates(cell);

    while (mask != 0) {
        // Isolate the lowest candidate bit and convert it to 1..9.
        int bit = mask & -mask;
        int digit = Integer.numberOfTrailingZeros(bit) + 1;
        board.place(cell, digit);
        if (solveMrv(board)) return true;
        board.remove(cell); // backtrack
        // Remove this candidate before trying the next one.
        mask ^= bit;
    }
    return false;
}

Each sketch uses the same four steps: choose the most constrained cell, iterate its candidate bit mask, recurse, and undo failed choices. Board helpers are omitted for clarity; the repository includes a complete C validator in tools/sudoku_validator.c.
