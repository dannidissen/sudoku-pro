Implementation sketch

Compare the browser implementation with the same MRV idea in modern C++.

function solveMRV(board) {
  if (board.isComplete()) return true;
  const cell = board.emptyCellWithFewestCandidates();
  for (const digit of cell.candidates) {
    board.place(cell, digit);
    if (solveMRV(board)) return true;
    board.remove(cell); // backtrack
  }
  return false;
}
#include <bit>
#include <cstdint>

bool solve_mrv(Board& board) {
    if (board.complete()) return true;
    Cell cell = board.fewest_candidates();
    uint16_t mask = board.candidates(cell);
    while (mask) {
        int digit = std::countr_zero(mask) + 1;
        board.place(cell, digit);
        if (solve_mrv(board)) return true;
        board.remove(cell); // backtrack
        mask &= mask - 1;
    }
    return false;
}
